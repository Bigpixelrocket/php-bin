# Shared extensions built beside the static s4 binary, one per line:
#
#   <extension> <on|off> [zend] [requires=<ext>[,<ext>...]] [pin=<source>:<url>]
#
# on        loaded by default through each install's bin/php.ini; the static
#           binary plus every "on" extension must equal expected-modules exactly
# off       shipped as lib/php/extensions/<extension>.so, commented out in php.ini
# zend      load with zend_extension= instead of extension=
# requires  other shared extensions that must load first
# pin       StaticPHP source name and the exact stable release archive to build,
#           so an upstream prerelease can never slip into a published build
#
# Deliberately absent: msgpack (building it compiles redis against it, so the
# default-on redis would stop loading without it), grpc (build time), swoole
# and swow (they replace the runtime model), parallel (needs a thread-safe
# build), and inotify (Linux only).

# Default on: the modules every release has always loaded.

imagick         on
mongodb         on
sqlsrv          on
pdo_sqlsrv      on
